================================================================================
VICEVERSE - JUDGE PORTAL: COMPLETE SUPABASE SETUP GUIDE & SQL SCRIPTS
================================================================================

This document contains step-by-step instructions to set up Supabase for the
ViceVerse Judge Portal, including project configuration, database schema,
realtime subscriptions, seed data, and authentication roles.

--------------------------------------------------------------------------------
STEP 1: CREATE SUPABASE PROJECT & GET API KEYS
--------------------------------------------------------------------------------
1. Go to https://supabase.com and sign in.
2. Click "New Project".
3. Enter:
   - Name: viceverse (or viceverse-judge)
   - Database Password: (save this securely)
   - Region: Choose the closest region (e.g., South Asia / Mumbai or Singapore)
4. Once created, go to: Project Settings -> API.
5. Copy the following keys to your `.env.local` in your Next.js project:
   - Project URL -> NEXT_PUBLIC_SUPABASE_URL
   - anon / public key -> NEXT_PUBLIC_SUPABASE_ANON_KEY
   - service_role key -> SUPABASE_SERVICE_ROLE_KEY (keep secret, server-only)

--------------------------------------------------------------------------------
STEP 2: RUN DATABASE SCHEMA IN SUPABASE SQL EDITOR
--------------------------------------------------------------------------------
1. In your Supabase Dashboard, click on "SQL Editor" in the left sidebar.
2. Click "New query".
3. Copy and paste the entire SQL block below, then click "RUN".

-- 1. Profiles Table (Tracks user roles, specifically 'judge', 'admin', etc.)
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email TEXT NOT NULL,
    full_name TEXT,
    role TEXT NOT NULL DEFAULT 'judge' CHECK (role IN ('judge', 'admin', 'lead', 'member')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Rounds Table (Hackathon evaluation rounds)
CREATE TABLE IF NOT EXISTS public.rounds (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    round_number INT NOT NULL UNIQUE,
    round_name TEXT NOT NULL,
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Teams Table (Hardcoded / Seeded Hackathon Teams)
CREATE TABLE IF NOT EXISTS public.teams (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    team_id TEXT NOT NULL UNIQUE, -- e.g., 'TEAM-101'
    team_name TEXT NOT NULL,
    domain TEXT NOT NULL, -- e.g., 'AI/ML', 'Web3', 'Healthcare'
    case_study TEXT, -- e.g., 'AI-driven Patient Triage System'
    canva_link TEXT, -- Canva presentation link
    drive_link TEXT, -- Google Drive prototype link
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Team Members Table (Members and their academic branches)
CREATE TABLE IF NOT EXISTS public.team_members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    team_id UUID NOT NULL REFERENCES public.teams(id) ON DELETE CASCADE,
    member_name TEXT NOT NULL,
    branch TEXT NOT NULL, -- e.g., 'CSE', 'ISE', 'ECE', 'AIML'
    is_lead BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. Round Teams Mapping (Allots teams to rounds and optionally specific judges)
CREATE TABLE IF NOT EXISTS public.round_teams (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    round_id UUID NOT NULL REFERENCES public.rounds(id) ON DELETE CASCADE,
    team_id UUID NOT NULL REFERENCES public.teams(id) ON DELETE CASCADE,
    assigned_judge_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(round_id, team_id)
);

-- 6. Evaluation Criteria Table (Admin-defined criteria per round)
CREATE TABLE IF NOT EXISTS public.evaluation_criteria (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    round_id UUID NOT NULL REFERENCES public.rounds(id) ON DELETE CASCADE,
    criteria_name TEXT NOT NULL,
    description TEXT,
    max_marks INT NOT NULL CHECK (max_marks > 0),
    display_order INT DEFAULT 1,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. Evaluations Table (Evaluation submissions by judges for teams in a round)
CREATE TABLE IF NOT EXISTS public.evaluations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    round_id UUID NOT NULL REFERENCES public.rounds(id) ON DELETE CASCADE,
    team_id UUID NOT NULL REFERENCES public.teams(id) ON DELETE CASCADE,
    judge_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    status TEXT NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'submitted')),
    total_score NUMERIC(5,2) DEFAULT 0,
    feedback TEXT,
    submitted_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(round_id, team_id, judge_id)
);

-- 8. Evaluation Scores Table (Individual criterion scores)
CREATE TABLE IF NOT EXISTS public.evaluation_scores (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    evaluation_id UUID NOT NULL REFERENCES public.evaluations(id) ON DELETE CASCADE,
    criteria_id UUID NOT NULL REFERENCES public.evaluation_criteria(id) ON DELETE CASCADE,
    score NUMERIC(5,2) NOT NULL DEFAULT 0 CHECK (score >= 0),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(evaluation_id, criteria_id)
);

-- 9. Trigger to automatically create a profile record when a new user signs up
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.profiles (id, email, full_name, role)
    VALUES (
        new.id,
        new.email,
        COALESCE(new.raw_user_meta_data->>'full_name', 'Judge'),
        COALESCE(new.raw_user_meta_data->>'role', 'judge')
    )
    ON CONFLICT (id) DO NOTHING;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
    AFTER INSERT ON auth.users
    FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

--- [END OF SQL SCHEMA SCRIPT] ---

--------------------------------------------------------------------------------
STEP 3: ENABLE REALTIME ON EVALUATIONS
--------------------------------------------------------------------------------
In the Supabase SQL Editor, run this command to enable Instant Realtime updates
for evaluation status changes on the dashboard:

ALTER PUBLICATION supabase_realtime ADD TABLE public.evaluations;

(You can also verify this in: Database -> Publications -> supabase_realtime -> toggle 'evaluations' ON).

--------------------------------------------------------------------------------
STEP 4: ROW LEVEL SECURITY (RLS) POLICIES
--------------------------------------------------------------------------------
Run the following SQL in SQL Editor to secure the database while letting judges
view teams and manage their evaluations:

ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.rounds ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.teams ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.team_members ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.round_teams ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.evaluation_criteria ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.evaluations ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.evaluation_scores ENABLE ROW LEVEL SECURITY;

-- Read policies for authenticated judges
CREATE POLICY "Allow authenticated read on profiles" ON public.profiles FOR SELECT TO authenticated USING (true);
CREATE POLICY "Allow authenticated read on rounds" ON public.rounds FOR SELECT TO authenticated USING (true);
CREATE POLICY "Allow authenticated read on teams" ON public.teams FOR SELECT TO authenticated USING (true);
CREATE POLICY "Allow authenticated read on team_members" ON public.team_members FOR SELECT TO authenticated USING (true);
CREATE POLICY "Allow authenticated read on round_teams" ON public.round_teams FOR SELECT TO authenticated USING (true);
CREATE POLICY "Allow authenticated read on criteria" ON public.evaluation_criteria FOR SELECT TO authenticated USING (true);

-- Evaluations policies for judges: Can read, insert, and update their own evaluations
CREATE POLICY "Judges can view all evaluations or own" ON public.evaluations FOR SELECT TO authenticated USING (true);

CREATE POLICY "Judges can insert own evaluation" ON public.evaluations FOR INSERT TO authenticated
WITH CHECK (auth.uid() = judge_id);

CREATE POLICY "Judges can update own draft evaluation" ON public.evaluations FOR UPDATE TO authenticated
USING (auth.uid() = judge_id AND status = 'draft')
WITH CHECK (auth.uid() = judge_id);

CREATE POLICY "Judges can manage evaluation scores" ON public.evaluation_scores FOR ALL TO authenticated
USING (
    EXISTS (
        SELECT 1 FROM public.evaluations
        WHERE evaluations.id = evaluation_scores.evaluation_id
        AND evaluations.judge_id = auth.uid()
        AND evaluations.status = 'draft'
    )
);

STEP 6: CREATE A JUDGE USER IN SUPABASE AUTH
--------------------------------------------------------------------------------
1. In Supabase Dashboard, go to Authentication -> Users.
2. Click "Add user" -> "Create user".
3. Enter Judge's Email (e.g., `judge@viceverse.com`) and Password.
4. Auto-confirm user: Yes.
5. In the SQL Editor, verify or set their role to 'judge':
   UPDATE public.profiles
   SET role = 'judge', full_name = 'Judge Panel 1'
   WHERE email = 'judge@viceverse.com';
